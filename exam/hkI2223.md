câu 1:[3đ]: viết hàm nhận vào 3 tham số ngày tháng năm cho biết đó là ngày thứ mấy trong tuần.
*write a function that takes in 3 arguments the date of the year indicating what day of the week it is.*
***Hướng làm: nếu không sử dụng các hàm trong thư viện có sẵn (vd: hàm Datetime,..) 
có thể sử dụng công thức Zeller's Congruence***
**bài làm:** 

```.c
#include <stdio.h>

// Hàm kiểm tra năm nhuận
int laNamNhuan(int nam) {
    return (nam % 4 == 0 && nam % 100 != 0) || (nam % 400 == 0);
}

// Hàm trả về số ngày trong tháng
int soNgayTrongThang(int thang, int nam) {
    // Kiểm tra tháng hợp lệ
    if (thang < 1 || thang > 12) {
        return -1; // Trả về -1 nếu tháng không hợp lệ
    }

    // Xử lý các tháng
    if (thang == 2) {
        return laNamNhuan(nam) ? 29 : 28; // Tháng 2: 29 nếu năm nhuận, 28 nếu không
    } else if (thang == 4 || thang == 6 || thang == 9 || thang == 11) {
        return 30; // Các tháng 4, 6, 9, 11 có 30 ngày
    } else {
        return 31; // Các tháng 1, 3, 5, 7, 8, 10, 12 có 31 ngày
    }
}

// Hàm main để kiểm tra
int main() {
    int thang, nam;

    // Nhập tháng và năm
    printf("Nhap thang: ");
    scanf("%d", &thang);
    printf("Nhap nam: ");
    scanf("%d", &nam);

    // Gọi hàm và in kết quả
    int soNgay = soNgayTrongThang(thang, nam);
    if (soNgay == -1) {
        printf("Thang khong hop le!\n");
    } else {
        printf("So ngay trong thang %d nam %d la: %d\n", thang, nam, soNgay);
    }

    return 0;
}
```

câu 2:4đ: viết hàm đếm số lần xuất hiện của phần tử nhỏ nhất trong mảng 1 chiều các số nguyên. 
vd: A:{1,2,10,3,1,9,1} -> số nhỏ nhất là 1 và xuất hiện 3 lần

``` c.
#include <stdio.h>

// Hàm đếm số lần xuất hiện của phần tử nhỏ nhất trong mảng
int demSoLanXuatHienMin(int a[], int n) {
    // Kiểm tra mảng rỗng
    if (n <= 0) {
        return 0;
    }

    // Tìm phần tử nhỏ nhất
    int min = a[0]; // Giả sử phần tử đầu tiên là nhỏ nhất
    for (int i = 1; i < n; i++) {
        if (a[i] < min) {
            min = a[i];
        }
    }

    // Đếm số lần xuất hiện của phần tử nhỏ nhất
    int dem = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == min) {
            dem++;
        }
    }

    return dem;
}

// Hàm main để kiểm tra
int main() {
    int n;

    // Nhập số phần tử của mảng
    printf("Nhap so phan tu cua mang: ");
    scanf("%d", &n);

    // Kiểm tra số phần tử hợp lệ
    if (n <= 0) {
        printf("Mang khong hop le!\n");
        return 0;
    }

    // Nhập mảng
    int a[n];
    printf("Nhap cac phan tu cua mang:\n");
    for (int i = 0; i < n; i++) {
        printf("a[%d] = ", i);
        scanf("%d", &a[i]);
    }

    // Gọi hàm và in kết quả
    int ketQua = demSoLanXuatHienMin(a, n);
    printf("So lan xuat hien cua phan tu nho nhat la: %d\n", ketQua);

    return 0;
}
```

câu 3:3đ: viết hàm tính trung bình cộng các số nguyên tố trong ma trận có các số nguyên. 
(số nguyên tố là số chỉ chia hết cho 1 và chính nó)

``` 
#include <stdio.h>

// Hàm kiểm tra số nguyên tố
int laSoNguyenTo(int n) {
    if (n < 2) {
        return 0; // Không phải số nguyên tố
    }
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            return 0; // Không phải số nguyên tố
        }
    }
    return 1; // Là số nguyên tố
}

// Hàm tính trung bình cộng các số nguyên tố trong ma trận
float trungBinhCongNguyenTo(int a[][100], int m, int n) {
    int tong = 0; // Tổng các số nguyên tố
    int dem = 0;  // Số lượng số nguyên tố

    // Duyệt ma trận
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            if (laSoNguyenTo(a[i][j])) {
                tong += a[i][j];
                dem++;
            }
        }
    }

    // Kiểm tra trường hợp không có số nguyên tố
    if (dem == 0) {
        return 0; // Trả về 0 nếu không có số nguyên tố
    }

    // Tính trung bình cộng
    return (float)tong / dem;
}

// Hàm main để kiểm tra
int main() {
    int m, n;

    // Nhập kích thước ma trận
    printf("Nhap so hang m: ");
    scanf("%d", &m);
    printf("Nhap so cot n: ");
    scanf("%d", &n);

    // Kiểm tra kích thước hợp lệ
    if (m <= 0 || n <= 0 || m > 100 || n > 100) {
        printf("Kich thuoc ma tran khong hop le!\n");
        return 0;
    }

    // Nhập ma trận
    int a[100][100];
    printf("Nhap cac phan tu cua ma tran:\n");
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            printf("a[%d][%d] = ", i, j);
            scanf("%d", &a[i][j]);
        }
    }

    // Gọi hàm và in kết quả
    float ketQua = trungBinhCongNguyenTo(a, m, n);
    if (ketQua == 0) {
        printf("Ma tran khong co so nguyen to!\n");
    } else {
        printf("Trung binh cong cac so nguyen to trong ma tran la: %.2f\n", ketQua);
    }

    return 0;
}
```
